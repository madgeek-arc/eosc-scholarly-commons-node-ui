server {
    ${ADD_SERVER_NAME}
    listen 80;
    root /usr/share/nginx/html;

    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/css text/javascript application/javascript application/json image/svg+xml;

    location / {
        try_files $uri$args $uri$args/ /index.html;
    }

    # Angular names its bundles <name>-<8 characters>.js|css (main-WYIZ6B7F.js), so a changed file always
    # gets a new URL and these can be cached for a year. index.html, ngsw.json and the service worker files
    # carry no hash and are not matched.
    location ~ "-[A-Z0-9]{8}\.(js|css)$" {
        expires 1y;
    }

    # The fonts and images that the stylesheets refer to are built into /app-media/ with the same kind of
    # hash in their names (the build option outputPath.media in angular.json). Not /media/: that path goes to
    # the Joomla site. The CORS header is the one the font rule below adds, which this prefix location overrides.
    location ^~ /app-media/ {
        expires 1y;
        add_header Access-Control-Allow-Origin *;
    }

    location ~* \.(eot|ttf|woff)$ {
        add_header Access-Control-Allow-Origin *;
    }

    ${ADD_PROXY_API}

    ${ADD_PROXY_PAGES}

    location ~ ^/(images|files|ebooks|pitch-decks)/ {
        root /usr/share/nginx/html/data;
    }

}
